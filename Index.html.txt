    <!-- Demo JavaScript (Testing Ke Liye) -->
    <script>
        function generateReview() {
            const input = document.getElementById('productInput').value;
            const loading = document.getElementById('loading');
            const resultBox = document.getElementById('resultBox');

            if (!input.trim()) {
                alert('Kripya kisi product ka naam likhein!');
                return;
            }

            // Demo UI Loading Toggle
            loading.classList.remove('hidden');
            resultBox.classList.add('hidden');

            // Simulating API Call (2 Seconds Delay)
            setTimeout(() => {
                document.getElementById('productTitle').innerText = input;
                loading.classList.add('hidden');
                resultBox.classList.remove('hidden');
            }, 1500);
        }
    </script>
